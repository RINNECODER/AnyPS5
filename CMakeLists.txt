cmake_minimum_required(VERSION 3.20)
project(Relinker CXX)

set(ANYPS5_TEST_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tests")

function(add_test_executable target)
    add_executable(${target} ${ARGN})
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "$<1:${ANYPS5_TEST_OUTPUT_DIRECTORY}>")
endfunction()

option(BUILD_TESTING "Build and register relinker tests" OFF)
if(BUILD_TESTING)
    enable_testing()
    if(MINGW)
        get_filename_component(testCompilerDirectory "${CMAKE_CXX_COMPILER}" DIRECTORY)
        foreach(runtimeName IN ITEMS libgcc_s_seh-1.dll libstdc++-6.dll libwinpthread-1.dll)
            if(NOT EXISTS "${testCompilerDirectory}/${runtimeName}")
                message(FATAL_ERROR "Cannot locate MinGW runtime: ${runtimeName}")
            endif()
            file(COPY "${testCompilerDirectory}/${runtimeName}" DESTINATION "${ANYPS5_TEST_OUTPUT_DIRECTORY}")
        endforeach()
    endif()
endif()

option(ANYPS5_ENABLE_SPIRV_TOOLS "Enable SPIRV-Tools validation and optimization" OFF)
option(APS5_ENABLE_TIMING_LOG "Enable frame timing logging" OFF)
add_compile_definitions(APS5_ENABLE_TIMING_LOG=$<BOOL:${APS5_ENABLE_TIMING_LOG}>)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(ANYPS5_CPU_RUNTIME_ONLY "Build the embedded guest CPU runtime without graphics libraries" OFF)
if(ANYPS5_CPU_RUNTIME_ONLY)
    if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
        message(FATAL_ERROR "ANYPS5_CPU_RUNTIME_ONLY currently requires native Apple Silicon macOS")
    endif()
    if(CMAKE_OSX_ARCHITECTURES AND NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
        message(FATAL_ERROR "The CPU runtime requires CMAKE_OSX_ARCHITECTURES=arm64")
    endif()
    add_subdirectory(core/cpu)
    return()
endif()

option(ANYPS5_METAL_UTILITIES_ONLY "Build only the native macOS Metal backend components" OFF)
if(ANYPS5_METAL_UTILITIES_ONLY)
    if(NOT APPLE)
        message(FATAL_ERROR "ANYPS5_METAL_UTILITIES_ONLY requires macOS")
    endif()
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 20)
    set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
    option(ANYPS5_METAL_SHADER_BRIDGE "Build the standalone SPIR-V to Metal shader bridge" OFF)
    option(ANYPS5_METAL_GUEST_REPLAY "Build the existing guest shader frontend replay on Metal" OFF)
    if(ANYPS5_METAL_GUEST_REPLAY AND NOT ANYPS5_METAL_SHADER_BRIDGE)
        message(FATAL_ERROR "ANYPS5_METAL_GUEST_REPLAY requires ANYPS5_METAL_SHADER_BRIDGE=ON")
    endif()
    if(ANYPS5_METAL_SHADER_BRIDGE)
        add_subdirectory(core/shader/recompiler/MetalBackend)
    endif()
    add_subdirectory(core/libs/prx/libSceAgcDriver/Graphics/Metal)
    if(ANYPS5_METAL_GUEST_REPLAY)
        add_subdirectory(core/shader/recompiler/MetalReplay)
    endif()
    return()
endif()
