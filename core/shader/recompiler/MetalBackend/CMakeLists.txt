cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(AnyPS5MetalShaderBridge LANGUAGES CXX)
endif()

get_filename_component(ANYPS5_BRIDGE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
if(NOT TARGET spirv-cross-msl)
    set(SPIRV_CROSS_STATIC ON CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_CLI OFF CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_ENABLE_GLSL ON CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_ENABLE_MSL ON CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_SKIP_INSTALL ON CACHE BOOL "" FORCE)
    foreach(component HLSL CPP REFLECT C_API UTIL)
        set(SPIRV_CROSS_ENABLE_${component} OFF CACHE BOOL "" FORCE)
    endforeach()
    if(NOT EXISTS "${ANYPS5_BRIDGE_ROOT}/3rdparty/SPIRV-Cross/CMakeLists.txt")
        message(FATAL_ERROR "Initialize the pinned 3rdparty/SPIRV-Cross submodule")
    endif()
    add_subdirectory("${ANYPS5_BRIDGE_ROOT}/3rdparty/SPIRV-Cross" "${CMAKE_CURRENT_BINARY_DIR}/spirv-cross" EXCLUDE_FROM_ALL)
endif()
add_library(anyps5_metal_shader_bridge STATIC MetalShaderBridge.cpp MetalConversionDiskCache.cpp)
target_compile_features(anyps5_metal_shader_bridge PUBLIC cxx_std_20)
# The MSL disk cache is valid only for the bridge, recompiler-header and SPIRV-Cross sources that
# produced it: hash them into its directory name, and reconfigure when any of them changes.
file(GLOB ANYPS5_MSL_CACHE_SOURCES
    "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/*.hpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/../*.hpp"
    "${ANYPS5_BRIDGE_ROOT}/3rdparty/SPIRV-Cross/spirv_*.cpp" "${ANYPS5_BRIDGE_ROOT}/3rdparty/SPIRV-Cross/spirv_*.hpp")
list(SORT ANYPS5_MSL_CACHE_SOURCES)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${ANYPS5_MSL_CACHE_SOURCES})
set(ANYPS5_MSL_CACHE_HASHES "")
foreach(source IN LISTS ANYPS5_MSL_CACHE_SOURCES)
    file(SHA256 "${source}" hash)
    string(APPEND ANYPS5_MSL_CACHE_HASHES "${hash}")
endforeach()
string(SHA256 ANYPS5_MSL_CACHE_VERSION "${ANYPS5_MSL_CACHE_HASHES}")
string(SUBSTRING "${ANYPS5_MSL_CACHE_VERSION}" 0 16 ANYPS5_MSL_CACHE_VERSION)
target_compile_definitions(anyps5_metal_shader_bridge PUBLIC ANYPS5_MSL_CACHE_VERSION="${ANYPS5_MSL_CACHE_VERSION}")
target_include_directories(anyps5_metal_shader_bridge PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(anyps5_metal_shader_bridge PRIVATE spirv-cross-msl compression)

option(ANYPS5_BUILD_METAL_SHADER_BRIDGE_TESTS "Shader bridge metadata and rejection contract test" OFF)
if(ANYPS5_BUILD_METAL_SHADER_BRIDGE_TESTS)
    enable_testing()
    add_executable(anyps5_metal_shader_bridge_test MetalShaderBridgeTest.cpp)
    target_compile_features(anyps5_metal_shader_bridge_test PRIVATE cxx_std_20)
    target_include_directories(anyps5_metal_shader_bridge_test PRIVATE "${ANYPS5_BRIDGE_ROOT}/3rdparty/SPIRV-Cross")
    target_link_libraries(anyps5_metal_shader_bridge_test PRIVATE anyps5_metal_shader_bridge)
    add_custom_target(anyps5_metal_shader_bridge_tests DEPENDS anyps5_metal_shader_bridge_test)
    add_test(NAME metal_shader_bridge_tests COMMAND anyps5_metal_shader_bridge_test)
    add_test(NAME anyps5_metal_shader_bridge_contract COMMAND anyps5_metal_shader_bridge_test)
    # The contract test converts shaders: keep it out of the user's shader cache.
    set_tests_properties(metal_shader_bridge_tests anyps5_metal_shader_bridge_contract PROPERTIES ENVIRONMENT ANYPS5_NO_SHADER_CACHE=1)
    add_executable(anyps5_metal_conversion_disk_cache_test MetalConversionDiskCacheTest.cpp)
    target_compile_features(anyps5_metal_conversion_disk_cache_test PRIVATE cxx_std_20)
    target_link_libraries(anyps5_metal_conversion_disk_cache_test PRIVATE anyps5_metal_shader_bridge)
    add_test(NAME anyps5_metal_conversion_disk_cache COMMAND anyps5_metal_conversion_disk_cache_test)
    add_executable(anyps5_shader_cache_eviction_test ShaderCacheEvictionTest.cpp)
    target_compile_features(anyps5_shader_cache_eviction_test PRIVATE cxx_std_20)
    add_test(NAME anyps5_shader_cache_eviction COMMAND anyps5_shader_cache_eviction_test)
endif()
