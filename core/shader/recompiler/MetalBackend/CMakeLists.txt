cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(AnyPS5MetalShaderBridge LANGUAGES CXX)
endif()

get_filename_component(ANYPS5_BRIDGE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
if(NOT TARGET spirv-cross-msl)
    set(SPIRV_CROSS_STATIC ON CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_CLI OFF CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_ENABLE_GLSL ON CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_ENABLE_MSL ON CACHE BOOL "" FORCE)
    set(SPIRV_CROSS_SKIP_INSTALL ON CACHE BOOL "" FORCE)
    foreach(component HLSL CPP REFLECT C_API UTIL)
        set(SPIRV_CROSS_ENABLE_${component} OFF CACHE BOOL "" FORCE)
    endforeach()
    if(NOT EXISTS "${ANYPS5_BRIDGE_ROOT}/3rdparty/SPIRV-Cross/CMakeLists.txt")
        message(FATAL_ERROR "Initialize the pinned 3rdparty/SPIRV-Cross submodule")
    endif()
    add_subdirectory("${ANYPS5_BRIDGE_ROOT}/3rdparty/SPIRV-Cross" "${CMAKE_CURRENT_BINARY_DIR}/spirv-cross" EXCLUDE_FROM_ALL)
endif()
add_library(anyps5_metal_shader_bridge STATIC MetalShaderBridge.cpp)
target_compile_features(anyps5_metal_shader_bridge PUBLIC cxx_std_20)
target_include_directories(anyps5_metal_shader_bridge PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(anyps5_metal_shader_bridge PRIVATE spirv-cross-msl)

option(ANYPS5_BUILD_METAL_SHADER_BRIDGE_TESTS "Shader bridge metadata and rejection contract test" OFF)
if(ANYPS5_BUILD_METAL_SHADER_BRIDGE_TESTS)
    enable_testing()
    add_executable(anyps5_metal_shader_bridge_test MetalShaderBridgeTest.cpp)
    target_compile_features(anyps5_metal_shader_bridge_test PRIVATE cxx_std_20)
    target_include_directories(anyps5_metal_shader_bridge_test PRIVATE "${ANYPS5_BRIDGE_ROOT}/3rdparty/SPIRV-Cross")
    target_link_libraries(anyps5_metal_shader_bridge_test PRIVATE anyps5_metal_shader_bridge)
    add_test(NAME anyps5_metal_shader_bridge_contract COMMAND anyps5_metal_shader_bridge_test)
endif()
